```Java
public Map<String, Object> execute(DispatchData dispatchData) {

    moduleResponseData.setPrice(0.0);

    moduleResponseData.setLimit(params.getLimit());
    moduleResponseData.setSearchMode(params.getSearchMode());
    moduleResponseData.setSimilarity(resultDTO.getSimilarity());
    moduleResponseData.setQuery(params.getUserChatInput());
    moduleResponseData.setSearchUsingReRank(resultDTO.getReRankerSwitch());
    moduleResponseData.setCharsLength(params.getUserChatInput().length());

    moduleResponseData.setQuoteQA(BeanUtil.copyToList(searchRes, SearchDataResponseItemType.class));

    Map<String, Object> map = new HashMap<>();
    map.put("isEmpty", searchRes.size() == 0 ? true : null);
    map.put("unEmpty", searchRes.size() > 0 ? true : null);

    map.put("responseData", moduleResponseData);
    return map;
```

```Java
// 第 113 行 ─── 三种 ID 获取模式
protected List<String> getKnowledgeIds(DatasetModule params, Map<String, Object> variables, List<String> datasetIds) {
    List<String> knowledgeIdList = new ArrayList<>();
    if (SU.isNotEmpty(variables)) {
        Object knowledgeIds = variables.get("knowledgeIds");
        if ((knowledgeIds instanceof String && StringUtils.isNotBlank((String) knowledgeIds)) || null != knowledgeIds) {
            CollectionUtil.addAll(knowledgeIdList, JSON.parseArray(JSON.toJSONString(knowledgeIds), String.class));
        }
        Object collectionIds = variables.get("collectionIds");
        if ((collectionIds instanceof String && StringUtils.isNotBlank((String) collectionIds)) || null != collectionIds) {
            CollectionUtil.addAll(datasetIds, JSON.parseArray(JSON.toJSONString(collectionIds), String.class));
        }
        if (SU.isNotEmpty(knowledgeIdList)) {
            return knowledgeIdList;
        }
    }

    List<String> knowledgeIdStringList = params.getDatasets();
    if (CollectionUtils.isEmpty(knowledgeIdStringList)) {
        return knowledgeIdList;
    }

    if (DYNAMIC_KNOWLEDGE_ID_SEARCH.getType().equals(params.getSearchType())) {
        List<DatasetModule.KnowledgeDatasetProps> kgdsidList = new ArrayList<>();
        for (String knowledgeIdString : knowledgeIdStringList) {
            List<DatasetModule.KnowledgeDatasetProps> result = parseIds(knowledgeIdString);
            CollectionUtil.addAll(kgdsidList, result);
        }
        kgdsidList = filterKnowledgeIds(kgdsidList);
        if (SizeUtil.isEmpty(kgdsidList)) {
            log.error("after filterKnowledgeIds(), no KnowledgeId remaining. original idString: '{}'", knowledgeIdStringList);
            throw new BusinessException(AgentFlowCommonErrorCode.KNOWLEDGE_NOT_EXIST);
        }
        for (DatasetModule.KnowledgeDatasetProps props : kgdsidList) {
            knowledgeIdList.add(props.getKnowledgeId());
            CollectionUtil.addAll(datasetIds, props.getDatasetIds());
        }
        // 数量限制：传入的知识库ID总量不超过50个，数据集ID总量不超过10000个。错误响应信息："超出限制：知识库 ID 最多50个，数据集 ID 最多10000个。"
        int knowledgeLimit = 50;
        int datasetLimit = 10000;
        if (SizeUtil.size(knowledgeIdList) > knowledgeLimit || SizeUtil.size(datasetIds) > datasetLimit) {
            log.warn("getKnowledgeIds(). knowledgeIdList.size/limit: '{}/{}', datasetIds.size/limit: '{}/{}'", SizeUtil.size(knowledgeIdList), knowledgeLimit, SizeUtil.size(datasetIds), datasetLimit);
            throw new BusinessException(AgentFlowCommonErrorCode.KNOWLEDGE_FLOW_DATASETSEARCH_COUNT_LIMIT);
        }
    } else if (KNOWLEDGE_SEARCH.getType().equals(params.getSearchType()) || REAL_CHAT_DOC_SEARCH.getType().equals(params.getSearchType())) {
        for (String dataset : knowledgeIdStringList) {
            try {
                DatasetModule.DatasetSearchProps datasetSearchProps = JSON.parseObject(dataset, DatasetModule.DatasetSearchProps.class);
                knowledgeIdList.add(datasetSearchProps.getDatasetId());
            } catch (Exception e) {
                log.error("getKnowledgeIds(). parse dataset error: '{}', dataset: '{}'", e, dataset, e);
            }
        }
    } else {
        log.error("getKnowledgeIds(). unknown searchType: '{}'", params.getSearchType());
    }

    log.info("DatasetSearchServiceImpl.getKnowledgeIds(): '({}){}..', datasetIds: '({}){}..'", SU.size(knowledgeIdList), FU.top(knowledgeIdList, 10), SU.size(datasetIds), FU.top(datasetIds, 10));
    return knowledgeIdList;
}
```

```Java
// 第 207 行 ─── 核心一致性过滤
private List<DatasetModule.KnowledgeDatasetProps> filterKnowledgeIds(List<DatasetModule.KnowledgeDatasetProps> knowledgeList) {
    // 去空
    knowledgeList = FU.toFilterList(knowledgeList, t -> StringUtils.isNotEmpty(t.getKnowledgeId()));
    Map<String, DatasetModule.KnowledgeDatasetProps> map = FU.toMap(knowledgeList, t -> t.getKnowledgeId());
    List<String> knowledgeIdList = new ArrayList<>(map.keySet());
    log.info("filterKnowledgeIds(). dynamic knowledge ids: '({}){}'", SizeUtil.size(knowledgeIdList), FU.top(knowledgeIdList, 10));

    // v6.2.2: 知识库列表必须：类型一致（文档、FAQ、Graph），向量索引模型一致（multilingual-e5-large、gte...）
    // 如果不一致的要么报错，要么取第一个知识库的类型(本次需求是这个策略)，并将后面的类型不一致的知识库id去掉
    // 过滤：限定只允许3种类型：文档知识库、FAQ 知识库、Graph 知识库
    List<Knowledge> knowledgeEntityList = knowledgesService.findByKnowledgeIds(knowledgeIdList);
    if (CollectionUtils.isEmpty(knowledgeEntityList)) {
        throw new BusinessException(AgentFlowCommonErrorCode.KNOWLEDGE_NOT_EXIST);
    }
    List<Knowledge> knowledge2 = knowledgeEntityList.stream().filter(t -> ALLOW_TYPES.contains(t.getType())).collect(Collectors.toList());
    if (knowledgeEntityList.size() != SizeUtil.size(knowledge2)) {
        log.info("filterKnowledgeIds(). allowTypes({}) filtered out from '{}' to '{}'", ALLOW_TYPES, knowledgeEntityList.size(), SizeUtil.size(knowledge2));
    }
    if (CollectionUtils.isEmpty(knowledge2)) {
        throw new BusinessException(AgentFlowCommonErrorCode.KNOWLEDGE_NOT_EXIST);
    }

    List<String> resultIdList = null;
    if (SizeUtil.onlyOne(knowledge2)) {
        resultIdList = FU.toList(knowledge2, Knowledge::getKnowledgeId);
    } else {
        // findByKnowledgeIds 顺序会被打乱，所以顺序需要以最初进入该方法的顺序为准
        Map<String, Knowledge> map1 = FU.toMap(knowledge2, t -> t.getKnowledgeId());
        Knowledge first = knowledgeList.stream()
                .map(t -> map1.get(t.getKnowledgeId()))
                .filter(Objects::nonNull)
                .findFirst().orElseGet(() -> knowledge2.get(0));
        if (null != first) {
            resultIdList = knowledge2.stream()
                    .filter(t -> Objects.equals(first.getType(), t.getType()) && Objects.equals(first.getVectorModel(), t.getVectorModel()))
                    .map(Knowledge::getKnowledgeId)
                    .collect(Collectors.toList());
        } else {
            log.error("filterKnowledgeIds(). first knowledge is null, knowledge2: '{}'", knowledge2);
        }
    }
    if (SizeUtil.size(resultIdList) != knowledge2.size()) {
        log.info("filterKnowledgeIds(). after alignment to first, filtered out: '{}', Remaining knowledge ids: '({}){}'", knowledge2.size() - SizeUtil.sizeZeroIfNull(resultIdList), SizeUtil.size(resultIdList), resultIdList);
    }
    return FU.toList(resultIdList, map::get);
}
```



**面试要点**: 多知识库联合检索时，必须保证所有知识库**类型一致**（都是文档/都是 FAQ/都是 Graph）且**向量模型一致**（都用 text\-embedding\-3\-large）。不一致的知识库会被静默丢弃。

```Java
// 第 254 行 ─── 三种检索模式分别构造参数
private KnowledgeRagDTO buildKnowledgeRagDTO(SessionUserInfo userInfo, DatasetModule dto, List<String> knowledgeIds, List<String> datasetIds, String chatId, String customerUserId, Boolean chatTestFlag) {
    CopyOptions copyOptions = CopyOptions.create().setIgnoreNullValue(true).setIgnoreError(true);

    KnowledgeRagDTO knowledgeRagDTO = BeanUtil.toBean(dto, KnowledgeRagDTO.class, copyOptions);
    knowledgeRagDTO.setKnowledgeIds(knowledgeIds);
    knowledgeRagDTO.setDatasetIds(datasetIds);
    knowledgeRagDTO.setQuery(dto.getUserChatInput());
    knowledgeRagDTO.setBusinessId(chatId);
    knowledgeRagDTO.setTenantId(userInfo.getTenantId());
    knowledgeRagDTO.setTeamId(String.valueOf(userInfo.getOrgId()));
    knowledgeRagDTO.setCustomerUserId(customerUserId);
    knowledgeRagDTO.setChatTestFlag(chatTestFlag);

    List<Integer> recallTypes = Optional.ofNullable(dto.getRecallTypes())
            .map(list -> list.stream()
                    .map(RecallTypeEnum::getIndexByCode).filter(Objects::nonNull).collect(Collectors.toList())
            ).orElse(null);
    knowledgeRagDTO.setRecallTypes(recallTypes);

    VectorSearchDTO vectorSearchDTO = BeanUtil.toBean(dto, VectorSearchDTO.class, copyOptions);
    FullTextSearchDTO fullTextSearchDTO = BeanUtil.toBean(dto, FullTextSearchDTO.class, copyOptions);

    switch (KnowledgeSearchModeEnum.getEnum(knowledgeRagDTO.getSearchMode())) {
        case EMBEDDING:
            knowledgeRagDTO.setVectorSearchDTO(vectorSearchDTO);
            break;
        case FULLTEXT_RECALL:
            knowledgeRagDTO.setFullTextSearchDTO(fullTextSearchDTO);
            break;
        case MIXED_RECALL:
            MixSearchDTO mixSearchDTO = BeanUtil.toBean(dto, MixSearchDTO.class, copyOptions);
            knowledgeRagDTO.setMixSearchDTO(mixSearchDTO);

            vectorSearchDTO.setRecallLimit((int) (knowledgeRagDTO.getRecallLimit() * mixSearchDTO.getVectorSearchRatio()));
            fullTextSearchDTO.setRecallLimit((int) (knowledgeRagDTO.getRecallLimit() * mixSearchDTO.getFullTextSearchRatio()));

            knowledgeRagDTO.setFullTextSearchDTO(fullTextSearchDTO);
            knowledgeRagDTO.setVectorSearchDTO(vectorSearchDTO);
            break;
        default:
            break;
    }

    return knowledgeRagDTO;
}
```

```Java
@Override
public KnowledgeSearchResultDTO knowledgeSearch(KnowledgeRagDTO dto) {

    if (!CollectionUtils.isEmpty(dto.getRecallConfigs())) {
        List<String> recallTypesOfHelp = dto.getRecallConfigs().stream()
                .filter(configDTO -> RecallTypeEnum.isHelped(configDTO.getRecallType()))
                .map(RecallConfigDTO::getRecallType)
                .collect(Collectors.toList());

        List<Integer> recallTypes = Optional.ofNullable(recallTypesOfHelp)
                .map(list -> list.stream()
                        .map(RecallTypeEnum::getIndexByCode).filter(Objects::nonNull).collect(Collectors.toList())
                ).orElse(null);
        dto.setRecallTypes(recallTypes);
    }

    long start = System.currentTimeMillis();

    validateParam(dto);

    log.info("knowledge search dto:{}", JSONObject.toJSONString(dto));

    boolean ragAuth = !KnowledgeSearchTypeEnum.REAL_CHAT_DOC_SEARCH.getType().equals(dto.getSearchType()) && StringUtils.isNotBlank(dto.getCustomerUserId());

    //数据集权限控制
    if (knowledgeAuthSearchSwitch && ragAuth) {
        List<String> authDatasetIds = getDatasetIdsByUserPermission(dto);

        if (CollectionUtils.isEmpty(authDatasetIds)) {
            double runningTime = new BigDecimal(System.currentTimeMillis() - start).divide(new BigDecimal(1000), 3, RoundingMode.HALF_UP).doubleValue();
            return KnowledgeSearchResultDTO.builder()
                    .searchMode(dto.getSearchMode())
                    .modelName(Optional.ofNullable(dto.getVectorSearchDTO()).map(VectorSearchDTO::getModelName).orElse(null))
                    .similarity(Optional.ofNullable(dto.getVectorSearchDTO()).map(VectorSearchDTO::getSimilarity).orElse(null))
                    .reRankerSwitch(Optional.ofNullable(dto.getMixSearchDTO()).map(MixSearchDTO::getReRankerSwitch).orElse(false))
                    .searchRes(Lists.newArrayList())
                    .limit(dto.getLimit())
                    .runningTime(runningTime)
                    .build();
        }

        log.info("fillDatasetIdsByUserPermission authDatasetIds:{}", authDatasetIds);
        dto.setDatasetIds(authDatasetIds);
    }

    // 知识库权限-添加知识库过滤，数据集权限过滤
    filterPermissionKnowledge(dto);

    //数据集启用禁用控制
    filterDatasetIdsByEnabled(dto);

    List<RecallResultDTO> recallResultDTOList = null;
    try {
        switch (KnowledgeSearchModeEnum.getEnum(dto.getSearchMode())) {
            case EMBEDDING:
                recallResultDTOList = vectorSearch(dto);
                break;
            case FULLTEXT_RECALL:
                recallResultDTOList = fullTextSearch(dto);
                break;
            case MIXED_RECALL:
                recallResultDTOList = mixSearch(dto);
                break;
            default:
                recallResultDTOList = Lists.newArrayList();
                break;
        }
    } catch (Exception e) {
        log.error("#knowledgeSearch() search error: '{}', dto: '{}'", e, JSON.toJSONString(dto), e);
        throw new BusinessException(AgentFlowCommonErrorCode.ES_VECTOR_SEARCH_PARAMETER_VALIDATION_EXCEPTION);
    }

    List<RecallResultDTO> resultDTOList = filterTokens(recallResultDTOList, dto.getLimit());

    List<RecallResultDTO> result = resultDTOList;

    //过滤掉禁用数据集
    if(dto.getSearchType().equals(KnowledgeSearchTypeEnum.KNOWLEDGE_SEARCH.getType())) {
        result = filterDisEnabledDatasets(resultDTOList);
    }

    //图片地址替换
    result.forEach(o->{
        String sourceText = o.getSourceText();
        if(StringUtils.isNotEmpty(sourceText) && sourceText.contains("s3://")){
            sourceText = sourceText.replaceAll("s3://", config.getHostWebUrl().concat("/resource?filePath=s3://"));
            o.setSourceText(sourceText);
        }
    });

    long end = System.currentTimeMillis();

    double runningTime = new BigDecimal(end - start).divide(new BigDecimal(1000), 3, RoundingMode.HALF_UP).doubleValue();

    //是否测试对话 测试对话不记录召回数量
    if(dto.getChatTestFlag()== null || !dto.getChatTestFlag()){
        //异步记录数据的召回数量
        updateRecallCount(result);
    }


    return KnowledgeSearchResultDTO.builder()
            .searchMode(dto.getSearchMode())
            .modelName(Optional.ofNullable(dto.getVectorSearchDTO()).map(VectorSearchDTO::getModelName).orElse(null))
            .similarity(Optional.ofNullable(dto.getVectorSearchDTO()).map(VectorSearchDTO::getSimilarity).orElse(null))
            .reRankerSwitch(Optional.ofNullable(dto.getMixSearchDTO()).map(MixSearchDTO::getReRankerSwitch).orElse(false))
            .searchRes(result)
            .limit(dto.getLimit())
            .runningTime(runningTime)
            .build();
}
```

### 5\.2 mixSearch\(\) 混合召回



```Java
private List<RecallResultDTO> mixSearch(KnowledgeRagDTO dto) {
    List<RecallResultDTO> vectorSearchResult = vectorSearch(dto);

    List<RecallResultDTO> fullTextSearchResult = fullTextSearch(dto);

    MixSearchDTO mixSearchDTO = dto.getMixSearchDTO();

    List<RecallResultDTO> result;

    Boolean rrfSwitch = Optional.ofNullable(mixSearchDTO.getRrfSwitch()).orElse(false);
    if (rrfSwitch) {
        result = rrfRank(mixSearchDTO, vectorSearchResult, fullTextSearchResult);
    } else {
        result = merge(vectorSearchResult, fullTextSearchResult);
    }

    if (CollectionUtils.isEmpty(result)) {
        return Lists.newArrayList();
    }

    Boolean reRankerSwitch = Optional.ofNullable(mixSearchDTO.getReRankerSwitch()).orElse(false);
    if (reRankerSwitch) {
```

### 5\.4 reRanker\(\) 重排序



```Java
private List<RecallResultDTO> reRanker(String query, List<Integer> recallTypes, String reRankerServiceUniCode, Double similarity, List<RecallResultDTO> result) {
    List<String> contextList = result.stream().map(RecallResultDTO::getQ).collect(Collectors.toList());
    SessionUserInfo userInfo = SessionThreadLocalUtil.getCurrentUserInfo();

    ModelParamDTO modelParamDTO = new ModelParamDTO();
    modelParamDTO.setInput(contextList);
    modelParamDTO.setContent(query);
    modelParamDTO.setReRankerServiceUniCode(reRankerServiceUniCode);
    modelParamDTO.setUserInfo(userInfo);

    List<PegRankerPrefabricationOutPutItemVO> ranker = modelService.pegRanker(modelParamDTO);

    if (CollectionUtils.isEmpty(ranker)) {
        log.warn("reRanker result is null, reRankerServiceUniCode is:{}", reRankerServiceUniCode);
        return Lists.newArrayList();
    }


    List<PegRankerPrefabricationOutPutItemVO> filterRanker = ranker;

    filterRanker = ranker.stream().filter(e -> similarity <= e.getRanker())
            .sorted(Comparator.comparing(PegRankerPrefabricationOutPutItemVO::getRanker).reversed()).collect(Collectors.toList());

    if (CollectionUtils.isEmpty(filterRanker)) {
        return Lists.newArrayList();
    }

    List<RecallResultDTO> resultDTOList = Lists.newArrayList();
    filterRanker.forEach(reRanker -> {
        RecallResultDTO recallResultDTO = result.get(reRanker.getIndex());
        if (Objects.isNull(recallResultDTO)) {
            return;
        }

        double score = CalculateUtil.getDecimal(reRanker.getRanker());
        recallResultDTO.getScore().add(getRrfScoreItem(recallResultDTO.getScore().size(), score, KnowledgeDataSearchScoreTypeEnum.RE_RANKER.getValue()));
        resultDTOList.add(recallResultDTO);
    });

    return resultDTOList;
}
```

## 6\. RrfRankerUtil\.java 算法源码分析



**文件**: `.../common/utils/RrfRankerUtil.java`

**行数**: 61 行

**职责**: RRF \(Reciprocal Rank Fusion\) 算法实现



```Java
/**
 * rrf算法
 *
 * @author hang.gao
 **/
public class RrfRankerUtil {

    @Getter
    @AllArgsConstructor
    public static class RankResult {
        private List<String> docIds;
        private Double weight;
    }

    /**
     * 算法公式：
     * rrf = weight * 1/(k + rank)
     * <p>
     * weight：通道的权重值
     * rank：文档在当前通道中的排名（从1开始）
     * k：平滑系数（通常设为60）
     *
     * @param rankResults {@link RankResult}
     * @param k           平滑因子
     * @return Map<String, Double>
     */
    public static Map<String, Double> rankFusion(List<RankResult> rankResults, int k) {
        Map<String, Double> finalScores = Maps.newHashMap();

        for (RankResult result : rankResults) {
            List<String> docs = result.getDocIds();
            double weight = result.getWeight();

            IntStream.range(0, docs.size()).forEachOrdered(i -> {
                double score = weight * (1.0 / (k + i + 1));
                finalScores.merge(docs.get(i), CalculateUtil.getDecimal(score), Double::sum);
            });
        }

        return finalScores.entrySet().stream()
                .sorted(Map.Entry.<String, Double>comparingByValue().reversed())
                .collect(Collectors.toMap(
                        Map.Entry::getKey,
                        Map.Entry::getValue,
                        (e1, e2) -> e1,
                        LinkedHashMap::new
                ));
    }
}
```



**RRF 算法示例**:



```Plain Text
输入:
  向量检索结果 (weight=6):  [docA, docB, docC]  (排名1,2,3)
  全文检索结果 (weight=4):  [docB, docD, docA]  (排名1,2,3)

计算:
  docA: 6/(60+1) + 4/(60+3) = 0.0984 + 0.0635 = 0.1619
  docB: 6/(60+2) + 4/(60+1) = 0.0968 + 0.0656 = 0.1624  ← 最高
  docC: 6/(60+3) + 0         = 0.0952 + 0      = 0.0952
  docD: 0        + 4/(60+2)  = 0      + 0.0645 = 0.0645

输出:
  docB(0.1624) > docA(0.1619) > docC(0.0952) > docD(0.0645)
```



**面试要点**: RRF 为什么比加权平均好？

1. RRF 对**排名**进行融合，而非对原始分数融合

2. 不同检索器（BM25 和 Cosine）的分数分布差异巨大，无法直接加权

3. 排名是统一量纲，不受分数分布影响

4. k=60 平滑因子避免排名靠后的文档分数被幂律分布压扁

```Java
public KnowledgeDocDirectRecallVO directVectorSearch(KnowledgeDocVectorSearchDTO dto) {
    // 1) 构建过滤条件
    Query query = Query.of(q -> q.bool(b -> b
        .must(n -> n.terms(t -> t.field("datasetId")
            .terms(knowledge -> knowledge.value(knowledgeListValue))))
        .mustNot(n -> n.term(TermQuery.of(t ->
            t.field("isEnabled").value(1))))  // isEnabled ≠ 1 → 排除禁用数据集
    ));

    // 2) 构建 knn 查询
    String vectorFieldName = "vector_" + dto.getVectors().size();  // vector_768
    KnnQuery knnQuery = KnnQuery.of(m -> m
        .field(vectorFieldName)
        .queryVector(dto.getVectors())      // 768 维向量
        .numCandidates(dto.getNumCandidates())  // 粗排候选数
        .k(dto.getTopK())                   // 最终返回数
        .filter(query));                    // 过滤条件

    // 3) 执行 ES 查询
    // source 过滤掉 vector 字段 (不返回向量，减少传输)
    SearchResponse<JSONObject> searchResponse = elasticsearchClient.search(s -> s
        .index(indicesList)
        .source(source -> source.filter(e -> e.excludes(List.of(vectorFieldName))))
        .knn(knnQuery)
        .size(dto.getLimit()), JSONObject.class);

    return buildRecallResult(searchResponse);
}
```

```Java
@Transactional
public Boolean autoAgentPublish(AgentPublishDTO dto) {
    // ===== 1) 基础检查 =====
    Agent agent = agentExtMapper.findByAgentId(userInfo, dto.getAgentId());

    // ===== 2) 构建快照 =====
    AgentSnapshot snapshot = buildAutoAgentSnapshot(dto, agent, date, userId, userName);

    // ===== 3) ★ 覆盖发布 vs 新增发布 ★ =====
    if (dto.isCovered()) {
        // 覆盖发布: 覆盖指定版本
        Integer publishVersion = dto.getVersion();
        AgentSnapshot existSnapshot = agentSnapshotService.findByAgentIdAndVersion(agentId, publishVersion);
        // 将该版本的旧记录设为归档 (ARCHIVED)
        updateWrapper.set(snapshotStatus, ARCHIVED)
                     .eq(agentId).eq(version, publishVersion);
        agentSnapshotService.update(updateWrapper);
        snapshot.setVersion(publishVersion);  // 复用原版本号
        // 是否设为默认
        if (isSetDefault) {
            snapshot.setDefaultStatus(1);
            // 将原默认版本取消
            updateWrapper.set(defaultStatus, 0).eq(agentId).eq(defaultStatus, 1);
        }
    } else {
        // 新增发布: 版本号自增
        AgentSnapshot maxVersion = agentSnapshotService.findMaxVersion(agentId);
        if (maxVersion == null) {
            snapshot.setDefaultStatus(1);      // 首次发布 → 默认版本
            snapshot.setVersion(1);
        } else {
            snapshot.setVersion(maxVersion.getVersion() + 1);  // 版本号 +1
        }
        if (dto.isSetDefault()) {
            // 取消原默认版本 → 设置新默认版本
            updateWrapper.set(defaultStatus, 0).eq(agentId).eq(defaultStatus, 1);
            snapshot.setDefaultStatus(1);
        }
    }

    // ===== 4) 更新 Agent 主表 =====
    agent.setVersion(snapshot.getVersion());
    agent.setPublicStatus(ONLINE);      // ← 标记为已发布
    agentMapper.updateByPrimaryKeySelective(agent);

    // ===== 5) 异步 CMDB 同步 =====
    if (cmdbSyncEnabled) {
        CompletableFuture.runAsync(() -> asyncSyncAgentToCmdb(agent));
    }

    // ===== 6) 插入快照记录 =====
    snapshot.setModuleJson(getAutoModuleJson(agent));  // ★ 完整配置快照
    agentSnapshotMapper.insert(snapshot);

    // ===== 7) 保存发布参数 =====
    AgentParams params = agentParamsService.getByAgentId(agentId);
    if (params == null) {
        agentParamsService.save({agentId, publishParam: dto.getParam()});
    } else {
        agentParamsService.updateById({publishParam: dto.getParam()});
    }
    return true;
}
```

```Java
// 第 1261 行 ─── ★ DFS 循环检测 ★
private boolean hasCycleHelper(Set<String> visited, SessionUserInfo userInfo, AutoAgentDTO agentDTO) {
    if (Objects.isNull(agentDTO)) return false;

    String toolId = agentDTO.getAgentId();
    // ★ 已访问过 → 发现环! ★
    if (visited.contains(toolId)) {
        return true;
    }
    visited.add(toolId);                    // ← 标记访问

    // 遍历子 Agent 工具
    Map<String, List<AutoAgentToolsDTO>> tools = agentDTO.getTools();
    if (tools != null) {
        List<AutoAgentToolsDTO> subAgents = tools.getOrDefault("agentInfoList", emptyList());
        for (AutoAgentToolsDTO child : subAgents) {
            // 递归查询子 Agent 配置
            agentDTO = this.findByAgentId(userInfo, child.getId(), ...);
            if (hasCycleHelper(visited, userInfo, agentDTO)) {
                return true;               // ← 子链中发现环
            }
        }
    }

    visited.remove(toolId);                // ★ 回溯 ★
    return false;
}
```



**DFS 回溯理解**:

```Plain Text
AgentA → AgentB → AgentC → AgentA

调用栈:
hasCycleHelper(A, visited={})
  visited = {A}
  查 A 的子 Agent → [B]
    hasCycleHelper(B, visited={A})
      visited = {A, B}
      查 B 的子 Agent → [C]
        hasCycleHelper(C, visited={A, B})
          visited = {A, B, C}
          查 C 的子 Agent → [A]
            hasCycleHelper(A, visited={A, B, C})
              A in visited → return TRUE!  ← 发现环
```

```Java
public void deleteAutoAgent(String agentId) {

    List<AutoAgent> autoAgents = autoAgentMapper.findByAgentId(agentId);

                .collect(toList());
            tools.put("agentInfoList", filtered);
        }
        autoAgent.setTools(toJSON(tools));
    });
    this.saveOrUpdateBatch(autoAgents);

    // 2) 清理发布态: AgentSnapshot 表中引用此 agentId 的 moduleJson
    List<AgentSnapshot> snapshots = agentSnapshotMapper.selectAutoAgent(agentId);
    snapshots.forEach(snapshot -> {
        List<AutoAgentDTO> dtos = parseJSON(snapshot.getModuleJson());
        dtos.stream().map(ParentAutoAgentDTO::getTools).forEach(tools -> {
            if (tools.containsKey("agentInfoList")) {
                tools.put("agentInfoList",
                    tools.get("agentInfoList").stream()
                        .filter(dto -> !dto.getId().equals(agentId))
                        .collect(toList()));
            }
        });
        snapshot.setModuleJson(toJSON(dtos));
    });
    agentSnapshotService.saveOrUpdateBatch(snapshots);
}
```

```Java
    return workflowFuture
        .thenCombine(abpFuture, (w, a) -> { combined.addAll(w); combined.addAll(a); })
        .thenCombine(autoAgentFuture, (ca, aa) -> { ca.addAll(aa); return ca; })
        .thenApply(list → page(list, dto.getPageNum(), dto.getPageSize()))
        .join();
```

```Java
 * @param list
 * @param maxTokens
 * @return

private List<SearchDataResponseItemType> filterSearchResultsByMaxChars(List<SearchDataResponseItemType> list,
                                                                       Integer maxTokens) {
    if (CollectionUtils.isEmpty(list)) {
        return list;
    }
    List<SearchDataResponseItemType> result = new ArrayList<>();
    Integer totalTokens = 0;
    for (int i = 0; i < list.size(); i++) {
        SearchDataResponseItemType item = list.get(i);
        String q = item.getQ();
        String a = item.getA();
        // 引用第三方类库计算的
        totalTokens += chatContextFilter.countPromptTokens(q + a, "");
        result.add(item);
        if (totalTokens > maxTokens) {
            break;
        }
    }
    if (CollectionUtils.isEmpty(result)) {
        return list.subList(0, 1);
    }
    return result;
}
```

```Java
public LastSummaryResult getLastSummary(List<ChatItem> histories, String moduleType, String moduleId, boolean summary) {
    String lastSummary = "";
    LastSummaryResult lastSummaryResult = new LastSummaryResult();
    lastSummaryResult.setLastSummary(lastSummary);
    lastSummaryResult.setHistories(histories);
    // 未开启摘要
    if (CollectionUtils.isEmpty(histories) || !summary) {
        if (CollectionUtils.isEmpty(histories)) {
            lastSummaryResult.setHistories(new ArrayList<>());
        }
        return lastSummaryResult;
    }
    // 截取部分 histories 过滤
    Integer summaryIndex = null;
    // 遍历history 获取上一次摘要
    for (int i = 0; i < histories.size(); i++) {
        ChatItem item = histories.get(i);
        if (null != item.getModuleSummaryData()) {
            List<ModuleSummary> moduleSummaryList = JSONUtil.toList(JSONUtil.toJsonStr(item.getModuleSummaryData()),
                ModuleSummary.class);
            for (ModuleSummary moduleSummary : moduleSummaryList) {
                // 组件取自己的摘要
                if (!moduleType.equalsIgnoreCase(moduleSummary.getModuleType())
                    || !moduleId.equalsIgnoreCase(moduleSummary.getModuleId())) {
                    continue;
                }
                if (StringUtils.isNotBlank(moduleSummary.getSummary()) && StringUtils.isNotBlank(moduleSummary.getLastSummary())) {
                    lastSummary = moduleSummary.getSummary();
                    summaryIndex = i;
                } else if (StringUtils.isNotBlank(moduleSummary.getSummary())) {
                    lastSummary = moduleSummary.getSummary();
                    summaryIndex = i;
                } else if (StringUtils.isNotBlank(moduleSummary.getLastSummary())) {
                    lastSummary = moduleSummary.getLastSummary();
                }
            }
        }
    }
    if (StringUtils.isBlank(lastSummary)) {
        return lastSummaryResult;
    }
    // summaryIndex
    if (null != summaryIndex) {
        if (summaryIndex >= 1 && summaryIndex <= histories.size() - 1) {
            histories = histories.subList(summaryIndex - 1, histories.size());
        }
    }

    lastSummaryResult.setLastSummary(lastSummary);
    lastSummaryResult.setHistories(histories);
    return lastSummaryResult;
}
```



**面试要点**:

- 摘要匹配是 `moduleType + moduleId` 的 `equalsIgnoreCase` 双重判断——同一组件只复用自己上一次存的摘要，不与其他节点串扰

- 有三个分支: `summary` 和 `lastSummary` 都有 → 用 summary \+ 记录裁切点; 只有 `summary` → 用 summary \+ 记录裁切点; 只有 `lastSummary` → 仅用 lastSummary（不裁切，因为没存过语义摘要）

- 两个前置早退: 未开启摘要 / 历史为空 → 直接返回初始 result; 遍历完 `lastSummary` 仍为空 → 直接返回，**不裁切历史**

- 裁切是 `subList(summaryIndex - 1, size)`——保留摘要所在那一轮及之后，丢弃更早的原始对话

```Java
// 过滤对话文本
SummaryResult getSummaryResult(String systemPrompt, String question, boolean summary, String reqId, Integer maxInput,
                               List<ChatItem> chatHistories, ModelListVO modelVO, Float temperature, String lastSummary, SessionUserInfo userInfo) {

    List<ChatItem> messages = new ArrayList<>();
    if (StringUtils.isNotBlank(systemPrompt) || StringUtils.isNotBlank(lastSummary)) {
        ChatItem systemChatItem = new ChatItem();
        systemChatItem.setObj(ChatRoleEnum.SYSTEM.getValue());
        systemChatItem.setValue(systemPrompt);
        if (StringUtils.isNotBlank(lastSummary)) {
            systemChatItem.setValue(systemPrompt + "\n[history]\n" + lastSummary);
        } else {
            systemChatItem.setValue(systemPrompt);
        }
        messages.add(systemChatItem);
    }
    messages.addAll(chatHistories);
    ChatItem humanChatItem = null;
    if (StringUtils.isNotBlank(question)) {
        humanChatItem = new ChatItem();
        humanChatItem.setObj(ChatRoleEnum.HUMAN.getValue());
        humanChatItem.setValue(question);
        messages.add(humanChatItem);
    }
    // 超长校验
    ChatFilterResult filterResult = getFilterMessages(messages, modelVO, summary);

    boolean aiSummary = filterResult.isAiSummary();
    List<ChatItem> filterMessages = filterResult.getResult();

    SummaryResult summaryResult = new SummaryResult();
    summaryResult.setFilterMessages(filterMessages);

    // 需要summary
    if (aiSummary) {
        summaryResult = getAISummary(chatHistories, modelVO, summary, maxInput,
                temperature, reqId, lastSummary,userInfo);
        String summaryPrompt = summaryResult.getSummaryPrompt();
        messages.clear();
        ChatItem systemChatItem = new ChatItem();
        systemChatItem.setObj(ChatRoleEnum.SYSTEM.getValue());
        systemChatItem.setValue(systemPrompt + "\n[history]\n" + summaryPrompt);
        messages.add(systemChatItem);
        if (null != humanChatItem) {
            messages.add(humanChatItem);
        }
        // 获取摘要后再次校验长度
        filterResult = getFilterMessages(messages, modelVO, true);
        if (filterResult.isAiSummary()) {
            throw new BusinessException(AgentFlowCommonErrorCode.CHAT_OVER_TOKEN_LIMIT_WITH_SUMMARY);
        }
        filterMessages = filterResult.getResult();
        summaryResult.setFilterMessages(filterMessages);
    }
    return summaryResult;
}
```



**面试要点**:

- **三级超长策略的载体**: ① `getFilterMessages()` 用 `modelVO.getHyperParameter().getMaxInput()`（缺省 `chatSummaryConfig.getMaxToken()`）做 Token 校验 → ② 超限触发 `getAISummary()` LLM 压缩 → ③ 摘要替换 history 后**再调一次** `getFilterMessages(messages, modelVO, true)` 二次校验，仍超限 → 抛 `CHAT_OVER_TOKEN_LIMIT_WITH_SUMMARY`

- System 消息只在 `systemPrompt` 或 `lastSummary` 非空时才构建（避免空 System 消息占 token）；摘要存在时 System Prompt 变成 `systemPrompt + "\n[history]\n" + lastSummary`

- 摘要后 `messages.clear()` 只保留 System \+ 当前问题——原始历史被摘要完全替换，这是 token 可控的关键

- 摘要后的二次校验**强制传 ****`true`**（summary 开关强制打开），确保无论如何都要把历史压缩到不超限

### 4\.4 getAISummary\(\) 摘要的 LLM 调用



```Java
public SummaryResult getAISummary(List<ChatItem> chatHistories, ModelListVO modelVO,
                                   boolean isSummary, Integer maxToken, Float temperature,
                                   String reqId, String lastSummary, SessionUserInfo userInfo) {
    if (!isSummary) {
        throw new BusinessException(AgentFlowCommonErrorCode.CHAT_OVER_TOKEN_LIMIT_WITHOUT_SUMMARY);
    }
    String summary = "";
    Map<String, Object> obj = new HashMap<>();
    StringBuilder quoteText = new StringBuilder();
    if (CollectionUtils.isEmpty(chatHistories)) {
        throw new BusinessException(AgentFlowCommonErrorCode.CHAT_QUESTION_OVER_TOKEN_LIMIT);
    }
    for (int i = 0; i < chatHistories.size(); i++) {
        ChatItem item = chatHistories.get(i);
        quoteText.append(item.getObj()).append(": ").append(item.getValue());
        if (i != chatHistories.size() - 1) {
            quoteText.append("\n");
        }
    }
    obj.put("lastSummary", lastSummary);
    obj.put("history", quoteText.toString());

    String content = MessageFormatUtils.replaceVariable(AIChatConstants.AI_SUMMARY_PROMPT, obj);

    List<ChatCompletionContentPart> contentParts = MessageFormatUtils.formatStr2ChatContent(content);
    List<ChatCompletionContent> concatMessages = new ArrayList<>();
    if (!CollectionUtils.isEmpty(contentParts)) {
        ChatCompletionContent chatContent = new ChatCompletionContent();
        chatContent.setRole(ChatRoleEnum.HUMAN.getDesc());
        chatContent.setContent(contentParts);
        concatMessages.add(chatContent);
    }
    List<ModelOutputDTO.ChatChoiceDTO> data = aiApiService.create(modelVO, temperature, maxToken,
            false, concatMessages, reqId,userInfo);
    if (!CollectionUtils.isEmpty(data) && Objects.nonNull(data.get(0).getMessage())) {
        summary = data.get(0).getMessage().getContent();
        if (summary.contains("{") && summary.contains("}")) {
            summary = summary.substring(summary.indexOf("{"), summary.lastIndexOf("}") + 1);
            summary = summary.replaceAll("\n", "").replaceAll(" ", "");
        }
    }
    SummaryResult result = new SummaryResult();
    result.setSummaryPrompt(summary);
    result.setLastSummaryPrompt(lastSummary);
    result.setFilterMessages(Lists.newArrayList());
    return result;
}
```



**面试要点**:

- 历史拼接格式 `"{obj}: {value}\n"`——即 `"Human: xxx\nAI: xxx\nHuman: xxx..."`，用换行分隔，最后一条不加换行

- 摘要模板是 `AIChatConstants.AI_SUMMARY_PROMPT`，入参两个 key: `lastSummary`（上一次摘要，用于增量续接）\+ `history`（全量原始历史）

- 摘要用 **`AIApiService.create()`**** 同步调用**（`stream=false`），不需要流式，只要最终一段文本

- LLM 返回后 `substring(indexOf("{"), lastIndexOf("}")+1)` 提取 JSON 片段 \+ `replaceAll("\n","").replaceAll(" ","")` 去空白——因为模型可能用 Markdown 代码块包裹 JSON，需要剥壳

- 角色用 `ChatRoleEnum.HUMAN.getDesc()`（即 "user"），而不是 getValue\(\)（"Human"）——**摘要调用走 GPT 原生角色名**

- 前置保护: `!isSummary` → 抛 `CHAT_OVER_TOKEN_LIMIT_WITHOUT_SUMMARY`（没开摘要开关却超限 → 直接拒绝）；历史为空 → 抛 `CHAT_QUESTION_OVER_TOKEN_LIMIT`

```Java
        modelParamDTO.setTop_p(chatCompletion.getTopP());
        modelParamDTO.setFrequency_penalty(chatCompletion.getFrequencyPenalty());
        modelParamDTO.setPresence_penalty(chatCompletion.getPresencePenalty());
        modelParamDTO.setSeed(chatCompletion.getSeed());
        if (CollectionUtils.isEmpty(modelStop)) {
            modelParamDTO.setStop(List.of("<|im_end|>"));
        } else {
            modelParamDTO.setStop(modelStop);
        }
        List<ModelParamDTO.MessageItem> messageItems = messages.stream().map(m -> {
            ModelParamDTO.MessageItem item = new ModelParamDTO.MessageItem();
            item.setRole(m.getRole());
            ChatCompletionContentPart contentPart = m.getContent().stream()
                    .filter(c -> "text".equals(c.getType())).findFirst().get();
            item.setContent(contentPart.getText());
            return item;
        }).collect(Collectors.toList());
        modelParamDTO.setMessages(messageItems);

        return modelParamDTO;
    }

}
```



**面试要点**:

- `create()` 是摘要专用同步链路: 内部 new 一个 `ChatCompletion` 只塞 temperature/maxToken → `buildModelParamDTO` → `modelService.modelProcess()`。默认 stop token 是 `List.of("<|im_end|>")`（防止模型自己续写）

- `createNew()` 流式时走老接口 `modelService.modelProcessStream()`：**一次调用返回完整 answer 字符串**，再手工包成 `ChatChoiceDTO`——注意它拿不到 usage/token 计费信息

- `createNew2()` 流式时走新接口 `modelManageService.modelProcessStreamNew(modelParamDTO, moduleId)`：返回 `ModelOutputDTO`（含 usage 字段），`moduleId` 用于内部按节点维度归集 token 计费。源码里保留了被注释掉的旧实现（`modelProcessStream` → 手工包 res），说明是一次接口升级的痕迹

- **`buildModelParamDTO()`**** 是核心**（private static）: 把 `ModelListVO`（模型配置）\+ `ChatCompletion`（本次对话参数）拼成 `ModelParamDTO`。几个关键点:

    - `serviceAddressList.get(0).getAddress()` 取**第一个服务地址**（负载均衡/主备由外部保证）

    - `temperature = max(temperature, 0.01)` 下限钳制，防止 0 导致随机性完全消失

    - `messages.stream().filter(c -> "text".equals(c.getType())).findFirst().get()` —— 只取文本类型内容，过滤图片/音频等多模态 content part

    - `stop` 为空时默认 `List.of("<|im_end|>")`

**三条调用链路对比**:



|方法|流式|返回值|使用场景|
|---|---|---|---|
|`create()`|否|`List<ChatChoiceDTO>`|摘要生成 \(AISummaryService\)|
|`createNew()`|可选|`List<ChatChoiceDTO>`|旧版 chatNode|

## 6\. SSEUtils\.java 流式推送



```Java
public class SSEUtils {

    private static final Logger log = LoggerFactory.getLogger(SSEUtils.class);
    // timeout
    private static final Long DEFAULT_TIME_OUT = 10 * 60 * 1000L;

    private static final Map<String, SseEmitter> subscribeMap = new ConcurrentHashMap<>();

    /**
     * 添加订阅
     */
    public static SseEmitter addSub(String reqId) {
        if (null == reqId || "".equals(reqId)) {
            return null;
        }

        SseEmitter emitter = subscribeMap.get(reqId);
        if (null == emitter) {
            emitter = new SseEmitterUTF8(DEFAULT_TIME_OUT);

            emitter.onError((e) -> {
                // 注册超时回调，超时后触发
                //log.error("onError e:{},reqId:{}", e, reqId);
                closeSub(reqId);
            });

            emitter.onTimeout(() -> {
                // 注册超时回调，超时后触发
                //log.warn("onTimeout,reqId=" + reqId);
                closeSub(reqId);
            });

            emitter.onCompletion(() -> {
                // 注册完成回调，调用 emitter.complete() 触发
                //log.info("onCompletion,reqId=" + reqId);
                closeSub(reqId);
            });
            subscribeMap.put(reqId, emitter);
        }
        return emitter;
    }

    public static void pubMsg(String reqId, String msg) {
        SseEmitter emitter = subscribeMap.get(reqId);
        if (null != emitter) {
            try {
                // 更规范的消息结构看源码
                emitter.send(event().id(reqId).data(
                        AgentFlowCommonConstant.SSE_DELIMITER + msg + AgentFlowCommonConstant.SSE_DELIMITER
                        // 更规范的消息结构看源码
                ));
            } catch (Exception e) {
                throw new RuntimeException(e);
            }
        }
    }

    public static void pubMsg(String reqId, String event, String msg) {
        SseEmitter emitter = subscribeMap.get(reqId);
        if (null != emitter) {
            try {
                // 更规范的消息结构看源码
                emitter.send(event().name(" " + event).data(" " + msg));
            } catch (Exception e) {
                log.error("An exception occurred in the SSE message:{}", e.getMessage());
            }
        }
    }

    public static SseEmitter getEmitter(String reqId) {
        return subscribeMap.get(reqId);
    }

    // 关闭订阅
    public static void closeSub(String reqId) {
        SseEmitter emitter = subscribeMap.get(reqId);
        if (null != emitter) {
            try {
                emitter.complete();
            } catch (Exception e) {
                log.error("An exception occurred when closing the SSE client", e);
            } finally {
                subscribeMap.remove(reqId);
            }
        }
    }

    public static void complete(String reqId) {
        SseEmitter emitter = subscribeMap.get(reqId);
        if (emitter != null) {
            emitter.complete();
        }
    }
}
```



**面试要点**:

- 核心是 `reqId → SseEmitter` 的 `ConcurrentHashMap`，前端先 `GET /stream/{reqId}` 建立连接 → 后端 `pubMsg()` 推送。不用 WebSocket，因为只需单向 Server→Client 推送

- `addSub()` 的幂等设计: reqId 已存在则直接复用已有 emitter，不重复创建

- 三个回调 `onError/onTimeout/onCompletion` 全部收敛到 `closeSub(reqId)` —— `closeSub` 里 `emitter.complete()` \+ **finally 中 ****`subscribeMap.remove(reqId)`**，保证任何退出路径都不泄漏 Map 条目（否则 reqId 会无限增长）

- `pubMsg(reqId, msg)` 推送时用 `event().id(reqId)` 带 id \+ `SSE_DELIMITER` 包裹数据；`pubMsg(reqId, event, msg)` 用 `event().name(" " + event)` 指定事件名（如 `ANSWER` / `MODULESTATUS`），`" "` 前缀是历史遗留格式约定

- 超时 `DEFAULT_TIME_OUT = 10 * 60 * 1000L`（10 分钟），前端 10s 心跳空数据防 Nginx 空闲断连

- `SseEmitterUTF8` 是 `SseEmitter` 的子类——重写 setContentType 指定 UTF\-8，解决中文乱码

**面试要点**: SSE 是单向推送 \(Server → Client\)，通过 `reqId` 作为频道 ID。前端先 `GET /stream/{reqId}` 建立连接，后端在执行过程中调用 `pubMsg()` 实时推送 token 和节点状态。不使用 WebSocket 是因为只需要单向推送。

#### Q1: RAG 检索引擎是基于 Milvus 还是 ES？



**两者都用，但主力是 ES 8\.x**。



|对比|ES \(主力\)|Milvus \(辅助\)|
|---|---|---|

**选型考量**: ES 8\.x 原生支持 KNN 后，不需要额外部署 Milvus 就能实现向量\+全文混合检索。但保留了 Milvus 通道（通过 VectorSpi 抽象层），用于纯向量高吞吐场景。



<a id="q2"></a>
#### Q2: RRF 融合为什么比直接加权好？



**问题**: BM25 分数（0\-100\+，无上限）和 Cosine 相似度（0\-1）量纲完全不同。直接加权：BM25 为 15\.3 的 chunk 碾压 Cosine 为 0\.92 的 chunk——全文检索天然会主导排序。



**RRF 解决方案**: 对排名融合，而非分数融合。`RRF(d) = weight × 1/(k + rank)`。排名是无量纲的——向量检索排第 1 和全文检索排第 1 等价。k=60 平滑因子防止排名靠后的文档被幂律压扁。



**具体示例**:

```Plain Text
向量检索排名: docA=1, docB=2, docC=3  (weight=6)
全文检索排名: docB=1, docD=2, docA=3  (weight=4)

RRF 计算:
  docA: 6/(60+1) + 4/(60+3) = 0.0984 + 0.0635 = 0.1619
  docB: 6/(60+2) + 4/(60+1) = 0.0968 + 0.0656 = 0.1624  ← 最高
  docC: 6/(60+3) + 0         = 0.0952
  docD: 0        + 4/(60+2)  = 0.0645

结论: docB 在两个通道都排前列 → RRF 分最高。docA 只在向量排第一 → 不如 docB。
```

### Agent 管理模块



<a id="q4"></a>
#### Q4: Agent 怎么决定调用哪个工具？



**两条路径**:



**路径一（WorkFlow 模式 ****`agentFramework="flow"`****）**: 用户在画布上手动编排节点顺序——工具调用不是 LLM 决定的，是画布决定了"先 datasetSearchNode → 再 chatNode"。



**路径二（AutoAgent 模式 ****`agentFramework="PA"`****）**: LLM 在 PlanReAct 循环中自主决定。流程：

1. ADKExecutor 启动时遍历 `agentMeta.tools`（pluginInfoList/mcpInfoList/workflowInfoList/agentInfoList）→ 为每个工具创建 `CommonTool` 实例 → 注册到 ADK LlmAgent

2. LLM 收到用户问题 \+ System Prompt（含工具列表 FunctionDeclaration）→ 在 `/*ACTION*/` 阶段输出 `function_call(tool_name, args)`

3. `CommonTool.run_async()` → HTTP 调用 Java `/api/v1/tools/run` → Java 端按 `type` 字段分发执行 → 返回结果

4. LLM 收到 `function_response` → 在 `/*REASONING*/` 阶段分析结果 → 决定继续调工具还是输出 `/*FINAL_ANSWER*/`

<a id="q5"></a>
#### Q5: Agent 快照版本管理怎么做的？



- 每次发布生成不可变快照记录（`agent_snapshot` 表），字段级快照覆盖 6 个 JSON 字段：tools（工具配置）、knowledge（知识库配置）、modelInfo（模型参数）、longTermMemory（长期记忆配置）、variableList（全局变量列表）、recommends（推荐问题列表）

- 默认版本：`isSetDefault=true` 时先将旧的默认快照取消（`updateDefaultSnapshot()`），再将新快照设为默认

- 发布前校验：DFS 循环检测（`hasCycleHelper` 递归检查 Agent 嵌套引用，发现环抛异常阻止发布）\+ 知识库配置完整性（backupStrategy/showSource/retrieveMaxLength 不可为空）

- 草稿模式：`draftMode=true` 全链路透传，工具调用层根据此标记跳过真实支付/消息发送/数据修改

### Chat 对话模块



<a id="q6"></a>
#### Q6: Token 超限的完整处理链路是怎样的？



```Plain Text
1. 快速路径: rawTextLen < maxTokens × 0.5 → 直接跳过计算（ChatContextFilter）
   为什么是 0.5？字符数 ≈ α × token数，中文 α≈1.5、英文 α≈4。字符数远小于 token 时必定不超限

2. 精确截断（未开摘要）: JTokkit 逐条计算 token 数 → 从后往前累加至 maxTokens → 截断
   Human+AI 配对移除——截断掉 Human 消息的同时移除配对的 AI 回复
   截断后 chats 为空 → 抛 CHAT_OVER_TOKEN_LIMIT 异常

3. LLM 摘要压缩（开启摘要）: 不截断，调 LLM 将完整历史压缩
   输入: "Human: xxx\nAI: xxx\nHuman: xxx..." 全量拼接
   输出: {"summary": "用户咨询了产品A电池参数(已告知5000mAh)，接着询问价格"}
   摘要替换原始 history: systemPrompt + "\n[history]\n" + summary
   摘要存 ChatItem.moduleSummaryData，下次对话读取时直接复用（subList 裁切旧记录）

4. 摘要后仍超限: 抛 CHAT_OVER_TOKEN_LIMIT_WITH_SUMMARY → 拒绝继续
```



<a id="q7"></a>
#### Q7: 为什么 SS E 流式用 reqId 频道广播而非 WebSocket？



**场景**: 实时 AI 对话只需要 Server→Client 单向推送，不需要双向通信。



**实现**: `SSEUtils` 维护全局 `Map<reqId, SseEmitter>`。前端 `GET /api/v1/chat/stream/{reqId}` 创建订阅（超时 10 分钟），后端 `pubMsg(reqId, event, msg)` 推送。推送两种事件：

- `ANSWER`: LLM token 级回复文本

- `MODULESTATUS`: 节点执行状态（status=1 运行中, 2 完成, 3 异常，含执行详情）

心跳每 10 秒发空数据防止 Nginx 代理超时断连。工作流执行结束 → `emitter.complete()` → 从 Map 移除。



<a id="q8"></a>
#### Q8: LLM 调用的模型路由怎么做？



**通过 ****`ModelService.queryModelList()`**** 动态获取可用模型列表**:

1. 调用模型服务的 API 查询当前可用的 LLM 模型实例

2. 遍历返回列表，按 `serviceUniCode` 匹配用户配置的模型编码（如 `maip_gpt-4`）

3. 匹配成功 → 获取 `modelUrl`（API 端点）\+ `hyperParameter`（temperature/maxTokens/timeout）

4. 匹配失败 → 抛 `LLM_MODEL_CALL_OFFLINE_EXCEPTION`

**三条调用链路**:

|方法|流式|返回值|使用场景|
|---|---|---|---|
|`create()`|否|`List<ChatChoiceDTO>`|摘要生成（AISummaryService）|
|`createNew()`|可选|`List<ChatChoiceDTO>`|旧版 chatNode|
|`createNew2()`|可选|`ModelOutputDTO`（含 usage）|新版 chatNode（支持 token 计费）|



**问题**: 当前无模型熔断降级——LLM 调用失败 → 重试 3 次 → 抛异常。改进方向：使用项目已引入的 Sentinel 做 `@SentinelResource` 熔断，熔断后自动降级到备用模型或预设兜底回复。
